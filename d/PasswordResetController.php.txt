<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Mail;

/**
 * «نسيت الباسوورد» by email code (v54).
 *  1) POST /api/web/password/forgot {email}            → a 6-digit code is emailed (valid 15 min)
 *  2) POST /api/web/password/reset  {email, code, password} → new password, all old logins signed out
 * The answer to step 1 is the same whether the email exists or not, so nobody can probe for accounts.
 * The code is stored hashed; 5 wrong tries kill it.
 */
class PasswordResetController extends Controller
{
    private const TTL_MIN = 15;
    private const MAX_TRIES = 5;

    private function key(string $email): string
    {
        return 'pwreset:' . sha1(mb_strtolower(trim($email)));
    }

    public function forgot(Request $request)
    {
        $data = $request->validate(['email' => 'required|email|max:120']);
        $email = mb_strtolower(trim($data['email']));
        $generic = response()->json(['ok' => true, 'message' => 'لو الإيميل ده عليه حساب، هيوصلك كود خلال دقيقة. بص في الـ Spam كمان.']);

        $user = User::whereRaw('LOWER(email) = ?', [$email])->orderByDesc('id')->first();
        if (!$user) return $generic;

        // one code per email per minute
        $k = $this->key($email);
        $old = Cache::get($k);
        if ($old && now()->timestamp - ($old['at'] ?? 0) < 60) return $generic;

        $code = (string) random_int(100000, 999999);
        Cache::put($k, ['h' => Hash::make($code), 'uid' => $user->id, 'tries' => 0, 'at' => now()->timestamp], now()->addMinutes(self::TTL_MIN));

        $name = trim($user->first_name) ?: 'عميلنا';
        $html = '<div dir="rtl" style="font-family:Tahoma,Arial,sans-serif;font-size:16px;color:#0A4C55;max-width:480px;margin:auto">'
            . '<h2 style="margin:0 0 12px">بكام</h2>'
            . '<p>أهلاً ' . e($name) . '،</p>'
            . '<p>ده كود تغيير الباسوورد بتاعك:</p>'
            . '<p style="font-size:32px;font-weight:bold;letter-spacing:8px;text-align:center;background:#EAF4F5;border-radius:12px;padding:14px;direction:ltr">' . $code . '</p>'
            . '<p>الكود شغال ' . self::TTL_MIN . ' دقيقة. لو انت ما طلبتش تغيير الباسوورد، تجاهل الرسالة دي وحسابك في أمان.</p>'
            . '<p style="color:#6E8A8E;font-size:13px">www.bekameg.com</p></div>';
        try {
            Mail::html($html, function ($m) use ($user) {
                $m->to($user->email)->subject('كود تغيير الباسوورد — بكام');
            });
        } catch (\Throwable $e) {
            Log::error('password reset mail failed: ' . $e->getMessage());
            Cache::forget($k);
            return response()->json(['message' => 'مقدرناش نبعت الإيميل دلوقتي. جرّب كمان شوية أو كلّمنا واتساب.'], 503);
        }
        return $generic;
    }

    public function reset(Request $request)
    {
        $data = $request->validate([
            'email' => 'required|email|max:120',
            'code' => 'required|digits:6',
            'password' => 'required|string|min:6|max:120',
        ]);
        $k = $this->key($data['email']);
        $rec = Cache::get($k);
        $bad = response()->json(['message' => 'الكود غلط أو خلص وقته. اطلب كود جديد.'], 422);
        if (!$rec) return $bad;
        if (!Hash::check($data['code'], $rec['h'])) {
            $rec['tries'] = ($rec['tries'] ?? 0) + 1;
            if ($rec['tries'] >= self::MAX_TRIES) Cache::forget($k);
            else Cache::put($k, $rec, now()->addMinutes(self::TTL_MIN));
            return $bad;
        }
        Cache::forget($k);
        $user = User::find($rec['uid']);
        if (!$user) return $bad;
        $user->password = Hash::make($data['password']);
        $user->reset_password_code = null;
        $user->save();
        // sign out every old session/app login of this account
        try { $user->tokens()->update(['revoked' => true]); } catch (\Throwable $e) { Log::warning('token revoke: ' . $e->getMessage()); }
        return response()->json(['ok' => true, 'message' => 'الباسوورد اتغيّر. سجّل دخول بالباسوورد الجديد.']);
    }
}
